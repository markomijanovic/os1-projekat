UPUTSTVO: getThreadID, join, joinAll i prioritetne niti
==========================================================

Ovo je prakticno uputstvo sta treba dodati u projekat ako se trazi:

1) getThreadID
2) join za cekanje jedne niti
3) joinAll za cekanje sve dece trenutne niti
4) prioritetno biranje niti bez nezavrsene dece

Primeri ispod su prilagodjeni ovom projektu, gde su osnovni fajlovi:

1. Syscall brojevi
------------------

Koristi slobodne syscall brojeve izmedju thread i semaphore poziva:

- 0x14 -> thread_get_id
- 0x15 -> thread_join
- 0x16 -> thread_add_child
- 0x17 -> thread_join_all

Postojeci brojevi ostaju:

- 0x11 -> thread_create
- 0x12 -> thread_exit
- 0x13 -> thread_dispatch
- 0x21+ -> semafori

Dodaj getThreadID logiku:

uint64 TCB::getRunningThreadId()
{

Dodaj join za jednu konkretnu nit:

int TCB::joinThread(TCB* target)
{
    if (running == nullptr || target == nullptr || target == running)
    {
        return -1;
    }

    if (target->isFinished())
    {
        return 0;
    }

    if (target->joinSem == nullptr)
    {
        target->joinSem = new _sem(0);
        if (target->joinSem == nullptr)
        {
            return -1;
        }
    }

    target->joinWaitCount++;
    return target->joinSem->wait();
}

Dodaj registraciju dece:

int TCB::addChild(TCB* parent, TCB* child)
{
    if (parent == nullptr || child == nullptr || parent == child)
    {
        return -1;
    }

    if (child->parent != nullptr || child->getState() == FINISHED)
    {
        return -1;
    }

    if (parent->childrenSem == nullptr)
    {
        parent->childrenSem = new _sem(0);
        if (parent->childrenSem == nullptr)
        {
            return -1;
        }
    }

    child->parent = parent;
    parent->childCount++;
    return 0;
}

Dodaj joinAll:

int TCB::joinAllChildren()
{
    if (running == nullptr)
    {
        return -1;
    }

    if (running->childCount <= 0)
    {
        return 0;
    }

    if (running->childrenSem == nullptr)
    {
        running->childrenSem = new _sem(0);
        if (running->childrenSem == nullptr)
        {
            return -1;
        }
    }

    running->waitingForChildren = true;
    int status = running->childrenSem->wait();
    running->waitingForChildren = false;
    return status;
}

Dodaj funkciju koja se poziva kada nit zavrsi:

void TCB::notifyFinish(TCB* finishedThread)
{
    if (finishedThread == nullptr)
    {
        return;
    }

    if (finishedThread->joinSem != nullptr)
    {
        int waiters = finishedThread->joinWaitCount;
        finishedThread->joinWaitCount = 0;

        for (int i = 0; i < waiters; i++)
        {
            finishedThread->joinSem->signal();
        }
    }

    if (finishedThread->parent == nullptr)
    {
        return;
    }

    TCB* parent = finishedThread->parent;
    finishedThread->parent = nullptr;

    if (parent->childCount > 0)
    {
        parent->childCount--;
    }

    if (parent->childCount == 0 &&
        parent->waitingForChildren &&
        parent->childrenSem != nullptr)
    {
        parent->childrenSem->signal();
    }
}

U TCB::dispatch(), u grani gde je stara nit FINISHED, pozovi notifyFinish pre nego sto nit ode u zombie:

else if (old != nullptr && old->getState() == FINISHED)
{
    notifyFinish(old);
    zombie = old;
}

Napomena za join:

Ovo je minimalan join koji radi kada roditelj zove join dok TCB deteta jos postoji.
Postojeca baza brise zavrseni TCB kroz zombie mehanizam, zato je najbezbedniji obrazac:

child->start();
child->join();
delete child;

Ako hoces potpuno robustan join, onda treba promeniti zivotni vek TCB-a tako da se zavrseni TCB ne obrise pre nego sto vlasnik Thread objekta zavrsi join/delete.

5. Izmene u src/syscall_c.cpp
-----------------------------

Dodaj C wrapper za getThreadID:

uint64 thread_get_id()
{
    register uint64 a0 asm ("a0") = 0x14;
    __asm__ volatile ("ecall" : "+r"(a0) :: "memory");
    return a0;
}

Dodaj C wrapper za join:

int thread_join(thread_t handle)
{
    register uint64 a0 asm ("a0") = 0x15;
    register uint64 a1 asm ("a1") = (uint64)handle;
    __asm__ volatile ("ecall" : "+r"(a0) : "r"(a1) : "memory");
    return (int)a0;
}

Dodaj C wrapper za addChild:

int thread_add_child(thread_t child)
{
    register uint64 a0 asm ("a0") = 0x16;
    register uint64 a1 asm ("a1") = (uint64)child;
    __asm__ volatile ("ecall" : "+r"(a0) : "r"(a1) : "memory");
    return (int)a0;
}

Dodaj C wrapper za joinAll:

int thread_join_all()
{
    register uint64 a0 asm ("a0") = 0x17;
    __asm__ volatile ("ecall" : "+r"(a0) :: "memory");
    return (int)a0;
}

9. Prioritetno biranje niti bez dece
------------------------------------

U src/scheduler.cpp izmeni Scheduler::get().

Cilj:

- ako postoji korisnicka nit bez nezavrsene dece, biraj nju
- ako postoje samo niti sa decom, sistem ipak mora da nastavi
- idle/kernel nit ne sme stalno da preskace roditelja sa decom

Primer stabilne implementacije:

TCB *Scheduler::get()
{
    List<TCB> postponed;
    TCB* selected = nullptr;
    bool skippedUserWithChildren = false;

    while (TCB* candidate = readyCoroutineQueue.removeFirst())
    {
        if (!candidate->isKernelThread() && !candidate->hasUnfinishedChildren())
        {
            selected = candidate;
            break;
        }

        if (candidate->isKernelThread() && !skippedUserWithChildren)
        {
            selected = candidate;
            break;
        }

        if (!candidate->isKernelThread() && candidate->hasUnfinishedChildren())
        {
            skippedUserWithChildren = true;
        }

        postponed.addLast(candidate);
    }

    if (selected != nullptr)
    {
        while (TCB* candidate = postponed.removeFirst())
        {
            readyCoroutineQueue.addLast(candidate);
        }
        return selected;
    }

    selected = postponed.removeFirst();
    while (TCB* candidate = postponed.removeFirst())
    {
        readyCoroutineQueue.addLast(candidate);
    }

    return selected;
}

Zasto ovako:

- niti bez dece imaju prednost
- roditelj sa decom se privremeno odlozi ako iza njega postoji dete ili druga nit bez dece
- ako iza njega postoji samo idle/kernel nit, roditelj se ipak ne izgladnjuje

12. Najcesce greske
-------------------

1) addChild se pozove pre start().

Pogresno:

ThreadB* b = new ThreadB(0);
this->addChild(b);
b->start();

Ispravno:

ThreadB* b = new ThreadB(0);
b->start();
this->addChild(b);

Razlog: start() pravi TCB i tek tada myHandle nije nullptr.

2) Scheduler uvek bira idle jer idle nema decu.

Zato kernel niti moraju da se tretiraju posebno preko isKernelThread().

3) joinAll se implementira preko busy wait petlje.

To ne treba raditi. Roditelj treba da se blokira na semaforu, a poslednje dete ga budi signalom.

4) Zaboravi se notifyFinish u dispatch().

Ako se roditelj ne obavesti kada dete zavrsi, join/joinAll nikada nece da se odblokira.

13.5. Kako testirati join
-------------------------

Test za join treba da dokaze da roditelj ne nastavlja pre deteta.

Primer:

static volatile int childDone = 0;

class JoinChild : public Thread
{
protected:
    void run() override
    {
        printString("child started\n");
        Thread::dispatch();
        childDone = 1;
        printString("child finished\n");
    }
};

void joinTest()
{
    childDone = 0;

    JoinChild* child = new JoinChild();
    child->start();

    printString("parent before join\n");
    child->join();
    printString("parent after join\n");

    if (childDone == 1)
    {
        printString("[OK] join waited child\n");
    }
    else
    {
        printString("[FAIL] join returned too early\n");
    }

    delete child;
}

Ocekivan redosled:

parent before join
child started
child finished
parent after join
[OK] join waited child


13.6. Kako testirati joinAll
----------------------------

Test za joinAll treba da dokaze da roditelj ceka svu registrovanu decu.

Primer:

static volatile int finishedChildren = 0;

class JoinAllChild : public Thread
{
protected:
    void run() override
    {
        Thread::dispatch();
        finishedChildren++;
        printString("one child finished\n");
    }
};

void joinAllSmallTest()
{
    finishedChildren = 0;

    JoinAllChild* c1 = new JoinAllChild();
    JoinAllChild* c2 = new JoinAllChild();
    JoinAllChild* c3 = new JoinAllChild();

    c1->start();
    thread_add_child(c1->getMyHandle());

    c2->start();
    thread_add_child(c2->getMyHandle());

    c3->start();
    thread_add_child(c3->getMyHandle());

    printString("parent before joinAll\n");
    thread_join_all();
    printString("parent after joinAll\n");

    if (finishedChildren == 3)
    {
        printString("[OK] joinAll waited all children\n");
    }
    else
    {
        printString("[FAIL] joinAll returned too early\n");
    }

    delete c1;
    delete c2;
    delete c3;
}

Bitno:

- start() mora pre addChild()
- joinAll ceka samo decu koja su registrovana preko addChild/thread_add_child
- ako zaboravis addChild, joinAll nema sta da ceka


13.7. Kako testirati prioritetne niti
-------------------------------------

Ideja testa:

- roditelj A napravi decu B i C
- A pozove dispatch ili joinAll
- deca treba da dobiju sansu pre roditelja koji jos ima nezavrsenu decu

Primer:

class PriorityChild : public Thread
{
public:
    PriorityChild(const char* name) : Thread(), name(name) {}

protected:
    void run() override
    {
        printString(name);
        printString(" child running\n");
    }

private:
    const char* name;
};

void priorityTest()
{
    PriorityChild* b = new PriorityChild("B");
    PriorityChild* c = new PriorityChild("C");

    b->start();
    thread_add_child(b->getMyHandle());

    c->start();
    thread_add_child(c->getMyHandle());

    printString("A before dispatch/joinAll\n");
    thread_dispatch();
    printString("A after dispatch\n");

    thread_join_all();
    printString("A after joinAll\n");

    delete b;
    delete c;
}

Ne treba ocekivati savrseno isti redosled za svaki print ako ima vise dispatch-a i timer
prekida. Test treba da proveri glavnu osobinu: roditelj ne sme da proglasi kraj pre dece.

13.11. Kako znati da je test dobar
----------------------------------

Dobar test:

- ima jasan pocetak i kraj ispisa
